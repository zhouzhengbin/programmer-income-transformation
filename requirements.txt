# ==========================================
# 🐍 Python 云端图像处理核心逻辑
# ==========================================
class handler(BaseHTTPRequestHandler):

    def do_GET(self):
        """返回移动端 H5 界面"""
        self.send_response(200)
        self.send_header('Content-type', 'text/html; charset=utf-8')
        self.end_headers()
        self.wfile.write(HTML_PAGE.encode('utf-8'))

    def do_POST(self):
        """处理 H5 发送的图片数据"""
        try:
            self.send_response(200)
            self.send_header('Content-type', 'application/json; charset=utf-8')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()

            content_length = int(self.headers.get('Content-Length', 0))
            post_data = self.rfile.read(content_length)
            req_data = json.loads(post_data.decode('utf-8'))

            base64_str = req_data.get('image', '')
            watermark_text = req_data.get('text', '仅限办业务使用')
            position_choice = req_data.get('position', '身份证正面')

            # 解码前端传来的 Base64 图片
            image_data = base64.b64decode(base64_str)
            base_image = Image.open(io.BytesIO(image_data)).convert("RGBA")
            
            # 执行核心打水印算法
            out_image = self.apply_watermark(base_image, watermark_text, position_choice)

            # 编码回 Base64 发给前端
            buffered = io.BytesIO()
            out_image.save(buffered, format="JPEG", quality=90)
            result_base64 = base64.b64encode(buffered.getvalue()).decode('utf-8')

            self.wfile.write(json.dumps({"status": "success", "result": result_base64}).encode('utf-8'))

        except Exception as e:
            self.wfile.write(json.dumps({"status": "error", "message": str(e)}).encode('utf-8'))

    def apply_watermark(self, base_image, watermark_text, position_choice):
        """继承自原脚本的安全区域水印算法"""
        width, height = base_image.size

        # 定义水印的安全区域（使用相对比例）
        if position_choice == "身份证正面":
            safe_box = (int(width * 0.18), int(height * 0.52), int(width * 0.60), int(height * 0.85))
        else:
            safe_box = (int(width * 0.20), int(height * 0.35), int(width * 0.80), int(height * 0.65))

        box_width = safe_box[2] - safe_box[0]
        box_height = safe_box[3] - safe_box[1]

        watermark_layer = Image.new('RGBA', base_image.size, (255, 255, 255, 0))
        draw = ImageDraw.Draw(watermark_layer)

        # 获取云端中文字体
        font_path = self.get_cloud_font()
        font_size = int(height / 25)  # 适配移动端压缩后的图片比例
        font = ImageFont.truetype(font_path, font_size)

        # 调用原版算法中的换行处理
        wrapped_lines = self.wrap_text_by_char(watermark_text, font, box_width)
        
        line_height = font.getbbox("A")[3] - font.getbbox("A")[1] + 5 
        total_text_height = line_height * len(wrapped_lines)

        current_y = safe_box[1] + (box_height - total_text_height) / 2

        for line in wrapped_lines:
            line_width = font.getlength(line)
            current_x = safe_box[0] + (box_width - line_width) / 2
            # 绘制文字（半透明红色，更醒目）
            draw.text((current_x, current_y), line, font=font, fill=(215, 58, 73, 160))
            current_y += line_height

        out_image = Image.alpha_composite(base_image, watermark_layer).convert("RGB")
        return out_image

    def wrap_text_by_char(self, text, font, max_width):
        """原脚本中的按字符换行算法"""
        lines = []
        current_line = ""
        for char in text:
            if font.getlength(current_line + char) <= max_width:
                current_line += char
            else:
                lines.append(current_line)
                current_line = char
        lines.append(current_line)
        return lines

    def get_cloud_font(self):
        """
        核心突破：解决 Serverless 环境无中文字体导致乱码的问题。
        动态拉取开源中文字体存入云端临时目录 /tmp。
        """
        font_path = "/tmp/wqy-microhei.ttc"
        if not os.path.exists(font_path):
            try:
                # 从可靠的 CDN 下载文泉驿微米黑字体
                font_url = "https://cdn.jsdelivr.net/gh/StellarCN/scp_zh@master/fonts/SimHei.ttf"
                urllib.request.urlretrieve(font_url, font_path)
            except Exception:
                pass # 如果下载失败，系统将尝试回退 (可能会变成方块)
        return font_path if os.path.exists(font_path) else None

    def do_OPTIONS(self):
        self.send_response(200)
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'POST, GET, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.end_headers()
